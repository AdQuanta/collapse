                  basis=basis, dtype=np.float64)
Sz2 = hamiltonian([["z", [[1 / N_pixel, i] for i in range(N_pixel + 1, N_total)]]], [],
                  basis=basis, dtype=np.float64)

times = np.linspace(0, 10, 101)

# Qubit in |0>
psi0 = np.zeros(basis.Ns, dtype=np.complex128)
psi0[0] = 1.
